cmake_minimum_required(VERSION 3.22)
project(quest_usb_forwarder LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(OPENXR_DIR "${CMAKE_SOURCE_DIR}/third_party/openxr")
add_library(openxr_loader SHARED IMPORTED)
set_target_properties(openxr_loader PROPERTIES IMPORTED_LOCATION "${OPENXR_DIR}/jni/arm64-v8a/libopenxr_loader.so")
add_library(native_app_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
target_include_directories(native_app_glue PUBLIC "${ANDROID_NDK}/sources/android/native_app_glue")
add_library(quest_forwarder SHARED app/main.cpp)
target_include_directories(quest_forwarder PRIVATE "${OPENXR_DIR}/prefab/modules/headers/include")
target_compile_definitions(quest_forwarder PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES XR_USE_TIMESPEC)
target_compile_options(quest_forwarder PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
target_link_options(quest_forwarder PRIVATE "-Wl,-u,ANativeActivity_onCreate" "-Wl,-z,max-page-size=16384")
target_link_libraries(quest_forwarder PRIVATE native_app_glue openxr_loader android log EGL GLESv3)
